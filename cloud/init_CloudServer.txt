*** AWS Ubuntu 22.04LTS - t2.micro ***

---<NodeJS install>---
# Node.js 설치
sudo apt update
sudo apt install -y nodejs npm
node -v
npm -v
#Node.js 프로젝트 초기화 및 패키지 설치 (Node.js 앱이 있는 디렉터리로 이동해서)
npm init -y
npm install express cors dotenv mqtt

---<MQTT>---
# Mosquitto MQTT 브로커 설치 (로컬 MQTT 브로커 사용 시)
sudo apt update
sudo apt install -y mosquitto mosquitto-clients
# Mosquitto 자동 실행 설정:
sudo systemctl enable mosquitto
sudo systemctl start mosquitto
# 포트 1883이 열려 있어야 하므로 방화벽 설정도 확인:
sudo ufw allow 1883

---<PM2>---
# PM2로 백그라운드 실행 -> 서비스를 꺼지지 않게 유지하려면 pm2를 사용
sudo npm install -g pm2
pm2 start CloudServer.js --name server
# pm2 명령어
pm2 list 
pm2 log server
pm2 stop server
pm2 restart server

---<TimescaleDB설치>---
sudo apt update && sudo apt upgrade -y
sudo apt install -y curl gnupg lsb-release
curl -fsSL https://www.postgresql.org/media/keys/ACCC4CF8.asc | sudo gpg --dearmor -o /usr/share/keyrings/postgresql.gpg
echo "deb [signed-by=/usr/share/keyrings/postgresql.gpg] http://apt.postgresql.org/pub/repos/apt $(lsb_release -cs)-pgdg main" | \
  sudo tee /etc/apt/sources.list.d/pgdg.list

sudo apt update
sudo apt install -y postgresql-17

wget --quiet -O - https://packagecloud.io/timescale/timescaledb/gpgkey | sudo gpg --dearmor -o /usr/share/keyrings/timescale.gpg

echo "deb [signed-by=/usr/share/keyrings/timescale.gpg] https://packagecloud.io/timescale/timescaledb/ubuntu/ $(lsb_release -cs) main" | \
  sudo tee /etc/apt/sources.list.d/timescaledb.list

sudo apt update
sudo apt install -y timescaledb-2-postgresql-17
sudo timescaledb-tune
sudo systemctl restart postgresql
# PostgreSQL 접속:
sudo -u postgres psql
